#### 路由配置
```typescript
// src/router/index.ts 中的登录相关路由
{
  path: '/login',
  name: 'Login',
  component: () => import('@/views/user/Login.vue'),
  meta: { title: '登录' }
},
{
  path: '/user',
  name: 'UserCenter',
  component: () => import('@/views/user/Index.vue'),
  meta: { title: '个人中心', requiresAuth: true }
}
```

#### 路由守卫
```typescript
// src/router/index.ts 中的守卫逻辑
router.beforeEach((to, from, next) => {
  if (to.meta.requiresAuth && !isLoggedIn()) {
    next({ path: '/login', query: { redirect: to.fullPath } })
  } else {
    next()
  }
})
```

```
your-project/
├── src/

│   ├── api/

│   └── router/
```

#### 2. 安装依赖（如果缺少）
```bash
npm install axios pinia element-plus
npm install -D @types/node
```

#### 3. 配置环境变量
```bash
# .env.development
VITE_API_BASE_URL=http://localhost:7001

# .env.production
VITE_API_BASE_URL=https://api.wudong.com
```

#### 4. 修改API基础地址（如果不同）
```typescript
// src/utils/request.ts
const BASE_URL = import.meta.env.VITE_API_BASE_URL || 'http://localhost:7001'
```

#### 5. 在main.ts中注册Store
```typescript
import { createPinia } from 'pinia'

const app = createApp(App)
app.use(createPinia())
```

#### 6. 测试登录功能
- 访问 `/login` 路径
- 输入测试账号密码
- 检查是否能正常登录

## ⚠️ 需要根据项目调整的地方

### 1. API接口路径
```typescript
// 如果你们后端接口路径不同，需要修改：
// src/api/user.ts

// 我的路径
export function loginByPassword(username: string, password: string) {
  return request.post<LoginResponse>('/app/user/login/password', { username, password })
}

// 如果后端路径是 /api/login，改成：
export function loginByPassword(username: string, password: string) {
  return request.post<LoginResponse>('/api/login', { username, password })
}
```

### 2. 响应数据格式
```typescript
// 如果后端返回格式不同，需要修改：
// src/utils/request.ts

// 我的格式：{ code: 1000, message: '', data: {} }
if (response.data.code === 1000) {
  return response.data.data
}

// 如果是：{ success: true, data: {} }，改成：
if (response.data.success) {
  return response.data.data
}
```

### 3. Token字段名
```typescript
// 如果后端返回的token字段名不同：
// src/stores/user.ts

// 我的：{ token, refreshToken, user }
function login(tokenData: { token: string; refreshToken: string; user: UserInfo }) {
  // ...
}

// 如果是：{ accessToken, refreshToken, userInfo }，改成：
function login(tokenData: { accessToken: string; refreshToken: string; userInfo: UserInfo }) {
  token.value = tokenData.accessToken
  userInfo.value = tokenData.userInfo
  setToken(tokenData.accessToken)
  // ...
}
```

### 4. LocalStorage键名
```typescript
// 如果想用自己的项目名，修改：
// src/utils/auth.ts

const TOKEN_KEY = 'wudong_token'           // 改成 'your_project_token'
const REFRESH_TOKEN_KEY = 'wudong_refresh_token'  // 改成 'your_project_refresh_token'
const USER_INFO_KEY = 'wudong_user_info'   // 改成 'your_project_user_info'
```
